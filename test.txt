# 1. Запуск сервера
node main.js -h 127.0.0.1 -p 3000 -c ./cache

# 2. GET: картинки немає в кеші, береться з http.cat (очікуємо 200, X-Cache: MISS)
curl -i http://127.0.0.1:3000/200

# 3. GET повторно: тепер з кешу (очікуємо 200, X-Cache: HIT)
curl -i http://127.0.0.1:3000/200

# 4. GET неіснуючого коду (очікуємо 404)
curl -i http://127.0.0.1:3000/999

# 5. GET з неправильним шляхом (очікуємо 404)
curl -i http://127.0.0.1:3000/abc

# 6. PUT: записати свою картинку (очікуємо 201)
curl -i -X PUT --data-binary @my.jpg -H "Content-Type: image/jpeg" http://127.0.0.1:3000/404

# 7. GET після PUT: має повернутись ваша картинка (очікуємо 200, X-Cache: HIT)
curl -i http://127.0.0.1:3000/404 -o after_put.jpg

# 8. DELETE: видалити з кешу (очікуємо 200)
curl -i -X DELETE http://127.0.0.1:3000/404

# 9. DELETE повторно: вже немає (очікуємо 404)
curl -i -X DELETE http://127.0.0.1:3000/404

# 10. Недозволений метод (очікуємо 405 і заголовок Allow)
curl -i -X POST http://127.0.0.1:3000/404

#11. Команди для перегляду заголовків:
curl -s -D - -o /dev/null http://127.0.0.1:3000/404
curl -s -D - -o /dev/null http://127.0.0.1:3000/404
curl -i http://127.0.0.1:3000/999
